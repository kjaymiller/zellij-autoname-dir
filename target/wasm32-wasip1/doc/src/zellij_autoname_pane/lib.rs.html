6#[derive(Default)]
7struct AutoNamePane {
8    // pane_id -> last name we set, so we don't fight user-set names
9    last_set: HashMap<PaneId, String>,
10    // pane_id -> last cwd seen
11    cwds: HashMap<PaneId, PathBuf>,
12}
13
14impl ZellijPlugin for AutoNamePane {
15    fn load(&mut self, _config: BTreeMap<String, String>) {
16        request_permission(&[
17            PermissionType::ReadApplicationState,
18            PermissionType::ChangeApplicationState,
19        ]);
20        subscribe(&[
21            EventType::CwdChanged,
22            EventType::PaneUpdate,
23            EventType::PermissionRequestResult,
24        ]);
25    }
26
27    fn update(&mut self, event: Event) -> bool {
28        match event {
29            Event::CwdChanged(pane_id, new_cwd, _clients) => {
30                self.cwds.insert(pane_id, new_cwd.clone());
31                self.apply_name(pane_id, &new_cwd, None);
32            }
33            Event::PaneUpdate(manifest) => {
34                for (_tab, panes) in manifest.panes.iter() {
35                    for p in panes {
36                        if p.is_plugin {
37                            continue;
38                        }
39                        let pane_id = PaneId::Terminal(p.id);
40                        let cwd = self.cwds.get(&pane_id).cloned();
41                        let running = pane_running_file(&p.terminal_command);
42                        if let Some(cwd) = cwd {
43                            self.apply_name(pane_id, &cwd, running.as_deref());
44                        } else if let Some(name) = running {
45                            self.maybe_rename(pane_id, &name);
46                        }
47                    }
48                }
49            }
50            _ => {}
51        }
52        false
53    }
54
55    fn render(&mut self, _rows: usize, _cols: usize) {}
56}

58impl AutoNamePane {
59    fn apply_name(&mut self, pane_id: PaneId, cwd: &Path, running: Option<&str>) {
60        let base = cwd
61            .file_name()
62            .and_then(|s| s.to_str())
63            .unwrap_or("/")
64            .to_string();
65        let name = match running {
66            Some(cmd) => format!("{base} · {cmd}"),
67            None => base,
68        };
69        self.maybe_rename(pane_id, &name);
70    }
71
72    fn maybe_rename(&mut self, pane_id: PaneId, name: &str) {
73        if self.last_set.get(&pane_id).map(String::as_str) == Some(name) {
74            return;
75        }
76        rename_pane_with_id(pane_id, name);
77        self.last_set.insert(pane_id, name.to_string());
78    }
79}

81fn pane_running_file(command: &Option<String>) -> Option<String> {
82    let cmd = command.as_ref()?;
83    let mut parts = cmd.split_whitespace();
84    let prog = parts.next()?;
85    if let Some(arg) = parts.next() {
86        let p = Path::new(arg);
87        if let Some(name) = p.file_name().and_then(|s| s.to_str()) {
88            return Some(format!("{} {}", basename(prog), name));
89        }
90    }
91    Some(basename(prog).to_string())
92}
93
94fn basename(s: &str) -> &str {
95    Path::new(s)
96        .file_name()
97        .and_then(|s| s.to_str())
98        .unwrap_or(s)
99}
